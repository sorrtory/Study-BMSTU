\newcommand{\image}[1]{

	\centering
	\includegraphics[width=1\textwidth]{Images/#1}
\end{figure}
}

% \codefromfile {Имя файла}

\newcommand{\codefromfile}[2]{
\begin{code}
	\inputminted[breaklines=true, framesep=10pt, fontsize=\footnotesize, firstline=1, linenos=true]{#2}{Listings/#1}
\end{code}
}
